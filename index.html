import numpy as np
from PIL import Image

def extract_lsb_sten(img_path):
    img = Image.open(img_path).convert('RGB')
    arr = np.array(img).flatten()
    bits = arr & 1
    bytes_arr = np.packbits(bits)
    if bytes_arr[:4].tobytes() == b"STEN":
        length = int.from_bytes(bytes_arr[4:8], 'big')
        return bytes_arr[8:8+length].tobytes().decode('utf-8')
    return "No STEN signature found"
